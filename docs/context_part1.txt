## app/Controllers/Admin/ItemController.php
```php
<?php declare(strict_types=1);

namespace App\Controllers\Admin;

use App\Controllers\BaseController;
use App\Libraries\ExternalUrlGuard;
use App\Models\AuditLogModel;
use App\Models\CronLogModel;
use App\Models\ItemModel;
use App\Models\ItemRevisionModel;
use App\Models\WhitelistModel;

class ItemController extends BaseController
{
    /**
     * Affiche le tableau de bord de modération (Nouvelles cartes & propositions de modif).
     */
    public function pending()
    {
        helper('text');

        $itemModel = new ItemModel();
        $revisionModel = new ItemRevisionModel();

        $revisions = $revisionModel->getPendingRevisions();

        foreach ($revisions as &$revision) {
            $changes = [];
            $fieldsToCompare = [
                'titre' => 'Titre',
                'sous_categorie' => 'Sous-catégorie',
                'status' => 'Statut',
                'image' => 'Image / Couverture',
                'lien' => 'Lien de visionnage/lecture',
                'description' => 'Description',
                'saison' => 'Saison',
                'total_saisons' => 'Total saisons',
                'episode' => 'Épisode',
                'total_episodes' => 'Total épisodes',
                'date_sortie' => 'Date de sortie',
            ];

            foreach ($fieldsToCompare as $field => $label) {
                $oldValue = $revision['original_' . $field] ?? '';
                $newValue = $revision[$field] ?? '';

                if ('date_sortie' === $field && (!empty($oldValue) || !empty($newValue))) {
                    $oldValue = $oldValue ? substr((string) $oldValue, 0, 10) : '';
                    $newValue = $newValue ? substr((string) $newValue, 0, 10) : '';
                }

                if ((string) $oldValue !== (string) $newValue) {
                    $changes[] = [
                        'field' => $field, 'label' => $label,
                        'old' => $oldValue, 'new' => $newValue,
                    ];
                }
            }

            $revision['changes'] = $changes;
        }

        $data = ['pendingItems' => $itemModel->where('is_public', 2)->findAll(), 'pendingRevisions' => $revisions];

        return view('admin/items/pending', $data);
    }

    /**
     * Valide et passe en 'public' une carte proposée.
     *
     * @param mixed $id
     */
    public function approve($id)
    {
        $itemModel = new ItemModel();
        $item = $itemModel->find($id);

        if ($item) {
            if ((int) $item->id_division === ItemModel::PERSONAL_DIVISION_ID) {
                $itemModel->update($id, [
                    'is_public' => 0,
                    'status' => $item->status === 'Public' ? 'Aucun' : $item->status,
                ]);
                (new AuditLogModel())->logAction('Modération : Carte personnelle gardée privée', "La carte personnelle ID {$id} ('{$item->titre}') ne peut pas être publiée.");
                return redirect()->back()->with('error', 'Cette carte appartient à la division Personnelles et ne peut pas être publiée.');
            }

            $itemModel->update($id, [
                'is_public' => 1,
                'status'    => 'Public',
            ]);
            (new AuditLogModel())->logAction('Modération : Approbation Carte', "Le SuperAdmin a validé la nouvelle publication de la carte ID {$id} ('{$item->titre}').");
        }

        return redirect()->back()->with('message', 'Nouvelle carte validée ! Elle est désormais visible de tous avec le statut Public.');
    }

    /**
     * Refuse la publication d'une carte (elle redevient privée).
     *
     * @param mixed $id
     */
    public function reject($id)
    {
        $itemModel = new ItemModel();
        $item = $itemModel->find($id);

        if ($item) {
            $itemModel->update($id, [
                'is_public' => 0,
                'status'    => 'Aucun',
            ]);
            (new AuditLogModel())->logAction('Modération : Refus Carte', "Le SuperAdmin a refusé la publication de la carte ID {$id} ('{$item->titre}'). Rétrogradation en privée.");
        }

        return redirect()->back()->with('error', "Nouvelle carte refusée. Elle est repassée en privée pour l'utilisateur.");
    }

    /**
     * Approuve un brouillon et écrase les données de la carte publique.
     *
     * @param mixed $revisionId
     */
    public function approveRevision($revisionId)
    {
        $revisionModel = new ItemRevisionModel();
        $itemModel = new ItemModel();

        $revision = $revisionModel->find($revisionId);

        if ($revision && 'pending' === $revision['revision_status']) {
            $updateData = [
                'titre' => $revision['titre'], 'sous_categorie' => $revision['sous_categorie'],
                'status' => 'Public',
                'image' => $revision['image'],
                'lien' => $revision['lien'], 'description' => $revision['description'],
                'saison' => $revision['saison'], 'total_saisons' => $revision['total_saisons'],
                'episode' => $revision['episode'], 'total_episodes' => $revision['total_episodes'],
                'date_sortie' => $revision['date_sortie'],
            ];

            $itemModel->update($revision['original_item_id'], $updateData);
            $revisionModel->update($revisionId, ['revision_status' => 'approved']);

            (new AuditLogModel())->logAction('Modération : Approbation Draft', "Validation du Draft ID {$revisionId}. Les données de la carte publique ID {$revision['original_item_id']} ('{$revision['titre']}') ont été écrasées avec succès.");

            return redirect()->back()->with('message', 'La modification a été fusionnée et est maintenant en ligne.');
        }

        return redirect()->back()->with('error', 'Révision introuvable ou déjà traitée.');
    }

    /**
     * Rejette la demande de modification (Brouillon supprimé, original conservé).
     *
     * @param mixed $revisionId
     */
    public function rejectRevision($revisionId)
    {
        $revisionModel = new ItemRevisionModel();
        $revision = $revisionModel->find($revisionId);

        if ($revision) {
            $revisionModel->update($revisionId, ['revision_status' => 'rejected']);
            (new AuditLogModel())->logAction('Modération : Refus Draft', "Rejet du Draft ID {$revisionId} pour la carte ID {$revision['original_item_id']}. La version publique n'a pas été affectée.");
        }

        return redirect()->back()->with('error', 'La modification a été refusée.');
    }

    /**
     * Suppression autoritaire d'une carte par un modérateur.
     *
     * @param mixed $id
     */
    public function delete($id)
    {
        $itemModel = new ItemModel();
        $item = $itemModel->find($id);

        if ($item) {
            $cronLogModel = new CronLogModel();
            $associatedReports = $cronLogModel->whereIn('task_name', [CronLogModel::TASK_PENDING_USER_REPORT, CronLogModel::TASK_CONFIRMED_DEAD])
                ->where('item_id', $id)->findAll();
            foreach ($associatedReports as $report) {
                $host = $this->normalizeReportHost((string) ($report['url_testee'] ?? ''));
                $remainingCards = $host === null ? [] : array_values(array_filter(
                    $this->activeCardsForReportHost($host),
                    static fn ($card) => (int) $card->id !== (int) $id
                ));
                if ($remainingCards !== []) {
                    $cronLogModel->update($report['id'], [
                        'item_id' => $remainingCards[0]->id,
                        'titre' => $remainingCards[0]->titre,
                    ]);
                } else {
                    $cronLogModel->delete($report['id']);
                }
            }

            $itemModel->delete($id);
            $cronLogModel->where('item_id', $id)->delete();
            (new AuditLogModel())->logAction('Modération : Suppression Carte', "Suppression définitive de la carte ID {$id} ('{$item->titre}').");

            return redirect()->back()->with('message', "La carte '{$item->titre}' a été définitivement supprimée.");
        }

        return redirect()->back()->with('error', 'Carte introuvable.');
    }

    /**
     * Affiche la liste complète des liens morts, des signalements en attente
     * et des domaines déclarés comme valides (whitelist).
     */
    public function deadLinks()
    {
        $deadItems = db_connect()->table('item')
            ->select('id AS item_id, titre, lien AS url_testee')
            ->where('link_status', 'dead')
            ->where('deleted_at', null)
            ->get()->getResultArray();

        $domains = [];
        $normalizedDeadItems = [];
        foreach ($deadItems as $deadItem) {
            $domain = $this->normalizeReportHost((string) ($deadItem['url_testee'] ?? ''));
            if ($domain === null) {
                continue;
            }
            $domains[$domain] = $domain;
            $deadItem['url_testee'] = 'https://' . $domain . '/';
            $normalizedDeadItems[] = $deadItem;
        }
        $domains = array_values($domains);
        sort($domains);

        $whitelistModel = new WhitelistModel();
        $validDomains = [];
        foreach ($whitelistModel->getAllOrdered() as $entry) {
            $host = $this->normalizeReportHost((string) ($entry['domain'] ?? ''));
            if ($host === null) {
                continue;
            }

            $url = 'https://' . $host . '/';
            if (! ExternalUrlGuard::isPublicHttpUrl($url)) {
                continue;
            }

            $validDomains[] = [
                'id'           => (int) $entry['id'],
                'host'         => $host,
                'url'          => $url,
                'created_at'   => $entry['created_at'] ?? null,
                'is_protected' => $whitelistModel->isProtectionActive($entry['created_at'] ?? null),
                'expires_at'   => $whitelistModel->getProtectionExpiresAt($entry['created_at'] ?? null),
                'items'        => $this->activeCardsForReportHost($host),
            ];
        }

        return view('admin/items/dead_links', [
            'deadItems'      => $normalizedDeadItems,
            'pendingReports' => (new CronLogModel())->where('task_name', CronLogModel::TASK_PENDING_USER_REPORT)->orderBy('last_run', 'ASC')->findAll(),
            'domains'        => $domains,
            'validDomains'   => $validDomains,
        ]);
    }

    private function normalizeReportHost(string $url): ?string
    {
        if (! preg_match('~^https?://~i', $url) && ! str_starts_with($url, '//')) {
            $url = 'https://' . $url;
        } elseif (str_starts_with($url, '//')) {
            $url = 'https:' . $url;
        }
        $host = parse_url($url, PHP_URL_HOST);
        if (! is_string($host) || $host === '') {
            return null;
        }
        return (string) preg_replace('/^www\./i', '', strtolower(rtrim($host, '.')));
    }

    private function activeCardsForReportHost(string $host): array
    {
        $candidates = (new ItemModel())->getActiveNonPersonalCandidatesByHost($host);

        return array_values(array_filter(
            $candidates,
            fn ($item) => $this->normalizeReportHost((string) $item->lien) === $host
        ));
    }

    /** Admin confirmation promotes every card on the host to confirmed dead. */
    public function approveUserLinkReport(string $id)
    {
        if (! ctype_digit($id) || (int) $id < 1) {
            return redirect()->back()->with('error', 'Identifiant de signalement invalide.');
        }
        $id = (int) $id;
        $cronLogs = new CronLogModel();
        $report = $cronLogs->find($id);
        if (! $report || $report['task_name'] !== CronLogModel::TASK_PENDING_USER_REPORT) {
            return redirect()->back()->with('error', 'Ce signalement est introuvable ou déjà traité.');
        }

        $host = $this->normalizeReportHost((string) $report['url_testee']);
        if ($host === null) {
            return redirect()->back()->with('error', 'Le domaine du signalement est invalide.');
        }
        $db = db_connect();
        $db->transBegin();
        try {
            $cards = $this->activeCardsForReportHost($host);
            if ($cards === []) {
                throw new \RuntimeException('Aucune carte active ne correspond à ce domaine.');
            }
            $itemModel = new ItemModel();
            foreach ($cards as $card) {
                if (! $itemModel->update($card->id, ['link_status' => 'dead'])) {
                    throw new \RuntimeException('Impossible de confirmer toutes les cartes du domaine.');
                }
            }
            $cronLogs->update($id, [
                'task_name' => CronLogModel::TASK_CONFIRMED_DEAD,
                'url_testee' => 'https://' . $host . '/',
                'code_erreur' => 0,
            ]);
            (new AuditLogModel())->logAction('Signalement confirmé', "Domaine {$host} confirmé comme mort par l’administration (" . count($cards) . ' carte(s)).');
            if (! $db->transStatus()) {
                throw new \RuntimeException('La transaction de modération a échoué.');
            }
            $db->transCommit();
            return redirect()->back()->with('message', "Le domaine {$host} est confirmé comme mort.");
        } catch (\Throwable $e) {
            $db->transRollback();
            log_message('error', 'Échec de confirmation du signalement {id}: {message}', ['id' => $id, 'message' => $e->getMessage()]);
            return redirect()->back()->with('error', 'Le signalement n’a pas pu être confirmé.');
        }
    }

    /**
     * Retire un domaine déclaré comme valide (whitelist) et déclare toutes ses cartes actives comme mortes.
     */
    public function declareValidDomainDead(string $id)
    {
        if (! ctype_digit($id) || (int) $id < 1) {
            return redirect()->back()->with('error', 'Identifiant de domaine invalide.');
        }

        $id = (int) $id;
        $whitelistModel = new WhitelistModel();
        $entry = $whitelistModel->find($id);

        if (! $entry) {
            return redirect()->back()->with('error', 'Ce domaine valide est introuvable ou a déjà été retiré.');
        }

        $host = $this->normalizeReportHost((string) ($entry['domain'] ?? ''));
        if ($host === null) {
            return redirect()->back()->with('error', 'Le domaine enregistré est invalide.');
        }

        $db = db_connect();
        $db->transBegin();
        try {
            $cards = $this->activeCardsForReportHost($host);
            $itemModel = new ItemModel();
            $cronLogs = new CronLogModel();

            foreach ($cards as $card) {
                if (! $itemModel->update($card->id, ['link_status' => 'dead'])) {
                    throw new \RuntimeException('Impossible de passer toutes les cartes du domaine en lien mort.');
                }
            }

            $whitelistModel->delete($id);

            $existingLogs = $cronLogs->whereIn('task_name', [
                CronLogModel::TASK_PENDING_USER_REPORT,
                CronLogModel::TASK_CONFIRMED_DEAD,
            ])->findAll();

            foreach ($existingLogs as $log) {
                if ($this->normalizeReportHost((string) ($log['url_testee'] ?? '')) === $host) {
                    $cronLogs->delete($log['id']);
                }
            }

            if ($cards !== []) {
                $cronLogs->insert([
                    'item_id'     => $cards[0]->id,
                    'titre'       => $cards[0]->titre,
                    'url_testee'  => 'https://' . $host . '/',
                    'code_erreur' => 0,
                    'task_name'   => CronLogModel::TASK_CONFIRMED_DEAD,
                    'last_run'    => date('Y-m-d H:i:s'),
                ]);
            }

            $count = count($cards);
            (new AuditLogModel())->logAction(
                'Domaine déclaré mort',
                "Le domaine {$host} a été retiré des domaines valides et déclaré comme mort par l’administration ({$count} carte(s) impactée(s))."
            );

            if (! $db->transStatus()) {
                throw new \RuntimeException('La transaction de passage en lien mort a échoué.');
            }

            $db->transCommit();

            if ($count === 0) {
                return redirect()->back()->with('message', "Le domaine {$host} a été retiré des domaines valides (aucune carte active associée).");
            }

            return redirect()->back()->with('message', "Le domaine {$host} a été déclaré comme mort ({$count} carte(s) impactée(s)).");
        } catch (\Throwable $e) {
            $db->transRollback();
            log_message('error', 'Échec du passage en lien mort pour le domaine valide {id}: {message}', [
                'id'      => $id,
                'message' => $e->getMessage(),
            ]);
            return redirect()->back()->with('error', 'Impossible de déclarer ce domaine comme mort.');
        }
    }

    /**
     * Migration en masse : Remplace un ancien domaine par un nouveau dans tous les liens de la base.
     */
    public function bulkUpdateDomain()
    {
        if ($this->request->is('post')) {
            $oldDomain = rtrim(str_replace(['https://', 'http://'], '', (string) $this->request->getPost('old_domain')), '/');
            $newDomain = rtrim(str_replace(['https://', 'http://'], '', (string) $this->request->getPost('new_domain')), '/');

            if (empty($oldDomain) || empty($newDomain)) {
                return redirect()->back()->with('error', 'Les champs sont requis.');
            }

            $itemModel = new ItemModel();
            $itemsToUpdate = $itemModel->like('lien', $oldDomain)->findAll();
            $count = 0;

            foreach ($itemsToUpdate as $item) {
                $newLien = str_replace($oldDomain, $newDomain, $item->lien);
                $itemModel->update($item->id, ['lien' => $newLien, 'link_status' => 'ok']);
                (new CronLogModel())->where('item_id', $item->id)->delete();
                $count++;
            }

            if ($count > 0) {
                (new AuditLogModel())->logAction('Maintenance', "Migration de '{$oldDomain}' vers '{$newDomain}' sur {$count} carte(s).");
                return redirect()->back()->with('message', "Le domaine a été remplacé sur {$count} carte(s).");
            }

            return redirect()->back()->with('error', "Aucune carte avec le domaine '{$oldDomain}'.");
        }
        return redirect()->back();
    }

    /**
     * Marque un domaine comme faux positif, le met temporairement en whitelist et retire ses alertes.
     */
    public function falseNegativeDomain()
    {
        $url = $this->request->getGet('url');

        if (empty($url)) {
            return redirect()->back();
        }

        $cronLogModel = new CronLogModel();
        $itemModel = new ItemModel();
        $whitelistModel = new WhitelistModel();

        $host = $this->normalizeReportHost((string) $url) ?? '';
        $count = 0;
        if ($host !== '') {
            foreach ($this->activeCardsForReportHost($host) as $card) {
                if (in_array($card->link_status, ['dead', 'reported'], true)) {
                    $itemModel->update($card->id, ['link_status' => 'ok']);
                    ++$count;
                }
            }
        }

        if (!empty($host)) {
            $whitelistModel->upsertDomain($host);
        }

        $logs = $cronLogModel->where('task_name', CronLogModel::TASK_CONFIRMED_DEAD)->findAll();
        foreach ($logs as $log) {
            if ($this->normalizeReportHost((string) ($log['url_testee'] ?? '')) === $host) {
                $cronLogModel->delete($log['id']);
            }
        }

        (new AuditLogModel())->logAction('Faux Négatif', "L'alerte pour le domaine {$url} a été ignorée ({$count} cartes restaurées) et mise en whitelist temporaire pour 7 jours.");

        return redirect()->back()->with('message', "Alerte retirée. Le domaine est considéré comme fonctionnel et ne pourra pas être signalé pendant 7 jours ({$count} carte(s) restaurée(s)).");
    }
```

```php
            return (string) preg_replace('/^www\./i', '', strtolower(rtrim($host, '.')));
        };

        $logs = $db->table('cron_logs')->select('task_name, url_testee')
            ->whereIn('task_name', [CronLogModel::TASK_PENDING_USER_REPORT, CronLogModel::TASK_CONFIRMED_DEAD])
            ->get()->getResultArray();
        foreach ($logs as $log) {
            $host = $normalizeHost((string) ($log['url_testee'] ?? ''));
            if ($host === null) {
                continue;
            }
            $state = $log['task_name'] === CronLogModel::TASK_PENDING_USER_REPORT ? 'pending' : 'confirmed';
            if (!isset($states[$host]) || $state === 'confirmed') {
                $states[$host] = $state;
            }
        }

        $reportedItems = $db->table('item')->select('lien, link_status')
            ->where('deleted_at', null)
            ->whereIn('link_status', ['reported', 'dead'])
            ->get()->getResultArray();
        foreach ($reportedItems as $item) {
            $host = $normalizeHost((string) ($item['lien'] ?? ''));
            if ($host === null) {
                continue;
            }
            $state = $item['link_status'] === 'reported' ? 'pending' : 'confirmed';
            if (!isset($states[$host]) || $state === 'confirmed') {
                $states[$host] = $state;
            }
        }

        $whitelistCutoff = date('Y-m-d H:i:s', time() - (WhitelistModel::PROTECTION_DAYS * 86400));
        $whitelist = (new WhitelistModel())->where('created_at >=', $whitelistCutoff)->findAll();
        foreach ($whitelist as $entry) {
            $host = $normalizeHost((string) ($entry['domain'] ?? ''));
            if ($host !== null && !isset($states[$host])) {
                $states[$host] = 'whitelist';
            }
        }

        return $states;
    }

    /**
```
```php
     * Incrémente rapidement l'épisode depuis le dashboard et récupère le total d'épisodes
     * de la nouvelle saison via TMDB (avec fallback Jikan pour les animes).
     *
     * @param mixed $id
     */
    public function incrementEpisode($id)
    {
        $item = $this->model->find($id);

        if ($item) {
            if (!$this->canManageItem($item)) {
                if ($this->request->isAJAX()) {
                    return $this->response->setStatusCode(403)->setJSON([
                        'success' => false,
                        'error' => 'Vous ne pouvez pas modifier cette carte.',
                        'csrf_token' => csrf_hash(),
                    ]);
                }
                return redirect()->back()->with('error', 'Vous ne pouvez pas modifier cette carte.');
            }

            $newEpisode = (int) $item->episode + 1;
            $newSaison = (int) $item->saison;
            $totalEpisodes = (int) $item->total_episodes;
            $totalSaisons = (int) $item->total_saisons;
            $newTotalEpisodes = $totalEpisodes > 0 ? $totalEpisodes : null;
            $updateData = [];

            // Détection automatique du passage à la saison suivante ou de la fin
            if ($totalEpisodes > 0 && $newEpisode > $totalEpisodes) {
                if ($totalSaisons > 0 && $newSaison >= $totalSaisons) {
                    $updateData['status'] = 'Terminé';
                    $newEpisode = $totalEpisodes;
                    $updateData['episode'] = $newEpisode;
                    $logMessage = "Mise à jour de la carte ID {$id} ('{$item->titre}') : Statut passé à Terminé.";
                } else {
                    $newEpisode = 1;
                    ++$newSaison;
                    $newTotalEpisodes = null;
                    $updateData['saison'] = $newSaison;
                    $updateData['episode'] = $newEpisode;
                    $updateData['total_episodes'] = null;

                    $logMessage = "Mise à jour de la carte ID {$id} ('{$item->titre}') : Épisode passé à {$newEpisode} (Saison {$newSaison})";
                }
            } else {
                $updateData['episode'] = $newEpisode;
                $logMessage = "Mise à jour de la carte ID {$id} ('{$item->titre}') : Épisode passé à {$newEpisode}";
            }

            // Passage automatique de "À voir" à "En cours" dès le premier +1 (sauf si déjà passé en Terminé)
            if ('À voir' === $item->status && empty($updateData['status'])) {
                $updateData['status'] = 'En cours';
            }

            $this->model->update($id, $updateData);

            $wasComplete = 'Terminé' === $item->status;
            $justCompleted = ('Terminé' === ($updateData['status'] ?? null));

            if (!empty($updateData['saison'])) {
                $updatedItem = $this->model->find($id);
                if ($updatedItem) {
                    $syncData = $this->syncGlobalEpisodesWithTMDB($updatedItem, true);

                    if (empty($syncData['total_episodes'])) {
                        $jikanTotal = $this->fetchSeasonEpisodesFromJikan((string) $updatedItem->titre, $newSaison);
                        if (null !== $jikanTotal) {
                            $syncData['total_episodes'] = $jikanTotal;
                        }
                    }

                    if (!empty($syncData['total_episodes'])) {
                        $newTotalEpisodes = (int) $syncData['total_episodes'];
                    }

                    $this->model->update($id, $syncData);
                }
            } else {
                $newEpisodeGlobal = (int) ($item->episode_global ?? $item->episode ?? 0) + (($wasComplete || $justCompleted) ? 0 : 1);
                $totalGlobal = $item->total_episodes_global ?? $item->total_episodes;
                if (null !== $totalGlobal && (int) $totalGlobal > 0 && $newEpisodeGlobal > (int) $totalGlobal) {
                    $newEpisodeGlobal = (int) $totalGlobal;
                }

                $this->model->update($id, [
                    'episode_global' => $newEpisodeGlobal,
                    'total_episodes_global' => $totalGlobal,
                ]);
            }

            (new AuditLogModel())->logAction('Incrémentation Rapide', $logMessage);

            if ($this->request->isAJAX()) {
                $finalItem = $this->model->find($id);

                return $this->response->setJSON([
                    'success' => true,
                    'new_episode' => $newEpisode,
                    'new_saison' => $newSaison,
                    'new_total_episodes' => $newTotalEpisodes,
                    'status' => $updateData['status'] ?? $item->status,
                    'new_link' => $finalItem ? $finalItem->getFinalLink() : null,
                    'csrf_token' => csrf_hash(),
                ]);
            }
        }
        return redirect()->back();

    /**
     * Vérifie en direct via AJAX la disponibilité d'une vidéo/lecteur sur une URL ciblée.
     */
    public function checkDispo()
    {
        $urlCible = $this->request->getGet('urlCible');
        if (!is_string($urlCible) || !ExternalUrlGuard::isPublicHttpUrl($urlCible)) {
            return $this->response->setJSON(['success' => false, 'error' => 'URL invalide.']);
        }

        $urlParts = parse_url((string) $urlCible);
        $host = strtolower(rtrim((string) ($urlParts['host'] ?? ''), '.'));

        $cacheKey = 'parsing_sites_config';
        if (!$sites = cache($cacheKey)) {
            $sites = (new SiteConfigModel())->where('is_active', 1)->findAll();
            foreach ($sites as &$site) {
                $site['ind_page'] = json_decode($site['indicateurs_page_invalide'], true) ?? [];
                $site['ind_lect'] = json_decode($site['indicateurs_lecteur'], true) ?? [];
            }
            cache()->save($cacheKey, $sites, 43200);
        }

        $currentConfig = null;
        foreach ($sites as $config) {
            $configuredHost = strtolower(parse_url(str_contains($config['domain'], '://') ? $config['domain'] : 'https://' . $config['domain'], PHP_URL_HOST));
            if ($host === $configuredHost || str_ends_with($host, '.' . $configuredHost)) {
                $currentConfig = $config;
                break;
            }
        }

        if (!$currentConfig)
            return $this->response->setJSON(['success' => false, 'error' => 'Domaine non supporté.']);

        $matches = [];
        @preg_match($currentConfig['regex_episode'], (string) $urlCible, $matches);
        $episodeExtrait = $matches[1] ?? null;

        try {
            $client = Services::curlrequest([
                'timeout' => 3,
                'connect_timeout' => 2,
                'http_errors' => false,
                'allow_redirects' => false,
                'user_agent' => 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) CodeIgniter4/Checker'
            ]);

            $response = $client->get($urlCible);
            if ($response->getStatusCode() === 404) {
                return $this->response->setJSON(['success' => true, 'disponible' => false, 'details' => ['erreur' => 'Page 404']]);
            }

            $html = (string) $response->getBody();
            $estSurFicheAnime = false;
            $lecteurPresent = false;

            foreach ($currentConfig['ind_page'] as $indicator) {
                if (stripos($html, $indicator) !== false) {
                    $estSurFicheAnime = true;
                    break;
                }
            }

            if (!$estSurFicheAnime) {
                foreach ($currentConfig['ind_lect'] as $indicator) {
                    $indicatorFinal = $episodeExtrait ? str_replace('{ep}', (string) $episodeExtrait, $indicator) : $indicator;
                    if (stripos($html, $indicatorFinal) !== false) {
                        $lecteurPresent = true;
                        break;
                    }
                }
            }

            unset($html);

            return $this->response->setJSON([
                'success' => true,
                'disponible' => !$estSurFicheAnime && $lecteurPresent,
                'details' => ['estSurFicheAnime' => $estSurFicheAnime, 'lecteurPresent' => $lecteurPresent],
            ]);
        } catch (\Throwable $e) {
            return $this->response->setJSON(['success' => false, 'error' => 'Timeout']);
        }
    }

    /**
     * Interroge TMDB pour calculer les compteurs globaux (épisodes vus au total et total de la série)
     * et, si demandé, récupérer le nombre d'épisodes de la saison en cours.
     */
    private function syncGlobalEpisodesWithTMDB(Item $item, bool $updateSeasonTotal = false): array
    {
        $apiKey = env('TMDB_API_KEY') ?? 'ba55da0439797150ed58c4e524584823';
        $client = Services::curlrequest([
            'timeout' => 5,
            'http_errors' => false,
        ]);

        $episodeGlobal = (int) $item->episode;
        $totalEpisodesGlobal = null;
        $seasonTotalEpisodes = null;
        $currentSaison = (int) $item->saison;

        try {
            $url = 'https://api.themoviedb.org/3/search/multi?query=' . urlencode($item->titre) . "&api_key={$apiKey}&language=fr-FR";
            $response = $client->get($url);

            if (200 === $response->getStatusCode()) {
                $body = json_decode($response->getBody(), true);

                if (!empty($body['results'])) {
                    foreach ($body['results'] as $result) {
                        if (isset($result['media_type']) && 'tv' === $result['media_type'] && isset($result['id'])) {
                            $tvUrl = "https://api.themoviedb.org/3/tv/{$result['id']}?api_key={$apiKey}&language=fr-FR";
                            $tvResponse = $client->get($tvUrl);

                            if (200 === $tvResponse->getStatusCode()) {
                                $tvBody = json_decode($tvResponse->getBody(), true);
                                $totalEpisodesGlobal = $tvBody['number_of_episodes'] ?? null;
                                $tmdbTotalSaisons = $tvBody['number_of_seasons'] ?? 1;

                                // Si l'utilisateur est sur une saison supérieure à celle connue par TMDB (Cas des animes fusionnés comme Frieren)
                                if ($currentSaison > 1 && $currentSaison > $tmdbTotalSaisons && $totalEpisodesGlobal > 0) {
                                    $currentTotalEpisodes = (int) $item->total_episodes;

                                    if ($currentTotalEpisodes > 0 && $totalEpisodesGlobal >= $currentTotalEpisodes) {
                                        $previousEpisodes = $totalEpisodesGlobal - $currentTotalEpisodes;
                                        $episodeGlobal = $previousEpisodes + (int) $item->episode;
                                    } else {
                                        $jikanEps = $this->fetchSeasonEpisodesFromJikan((string) $item->titre, $currentSaison);
                                        if (null !== $jikanEps && $jikanEps > 0 && $totalEpisodesGlobal >= $jikanEps) {
                                            $previousEpisodes = $totalEpisodesGlobal - $jikanEps;
                                            $episodeGlobal = $previousEpisodes + (int) $item->episode;
                                            $seasonTotalEpisodes = $jikanEps;
                                            $item->total_episodes = $jikanEps;
                                        }
                                    }
                                    if ((int) $item->total_saisons < $currentSaison) {
                                        $item->total_saisons = $currentSaison;
                                    }
                                } else {
                                    if (isset($tvBody['seasons']) && is_array($tvBody['seasons'])) {
                                        $previousEpisodes = 0;
                                        foreach ($tvBody['seasons'] as $season) {
                                            $seasonNum = (int) ($season['season_number'] ?? -1);
                                            $epCount = (int) ($season['episode_count'] ?? 0);

                                            if ($seasonNum > 0 && $seasonNum < $currentSaison) {
                                                $previousEpisodes += $epCount;
                                            } elseif ($seasonNum === $currentSaison && $epCount > 0) {
                                                $seasonTotalEpisodes = $epCount;
                                            }
                                        }
                                        if ($currentSaison > 1) {
                                            $episodeGlobal = $previousEpisodes + (int) $item->episode;
                                        }
                                    }
                                }
                            }
                            break;
                        }
                    }
                }
            }
        } catch (\Exception $e) {
            // Silence
        }

        $result = [
            'episode_global' => $episodeGlobal,
            'total_episodes_global' => $totalEpisodesGlobal,
        ];

        if ($updateSeasonTotal && null !== $seasonTotalEpisodes) {
            $result['total_episodes'] = $seasonTotalEpisodes;
        }

        return $result;
    }

    /**
     * Normalize any card URL to its lowercase hostname.
     */
    private function reportedLinkHost(string $url): ?string
    {
        $url = trim($url);
        if ($url === '')
            return null;
        if (str_starts_with($url, '//'))
            $url = 'https:' . $url;
        elseif (!preg_match('~^https?://~i', $url))
            $url = 'https://' . $url;

        $host = parse_url($url, PHP_URL_HOST);
        if (!is_string($host) || $host === '')
            return null;
        $host = strtolower(rtrim($host, '.'));
        return (string) preg_replace('/^www\./i', '', $host);
    }

    /**
     * Check both historical reports and cards already marked dead for this host.
     */
    private function domainAlreadyReported(string $host): bool
    {
        $db = db_connect();
        $logs = $db
            ->table('cron_logs')
            ->select('url_testee')
            ->whereIn('task_name', [CronLogModel::TASK_CONFIRMED_DEAD, CronLogModel::TASK_PENDING_USER_REPORT])
            ->like('url_testee', $host)
            ->get()
            ->getResultArray();
        foreach ($logs as $log) {
            if ($this->reportedLinkHost((string) ($log['url_testee'] ?? '')) === $host)
                return true;
        }

        $deadItems = $db
            ->table('item')
            ->select('lien')
            ->where('deleted_at', null)
            ->whereIn('link_status', ['reported', 'dead'])
            ->like('lien', $host)
            ->get()
            ->getResultArray();
        foreach ($deadItems as $deadItem) {
            if ($this->reportedLinkHost((string) ($deadItem['lien'] ?? '')) === $host)
                return true;
        }

        return false;
    }

    /**
     * Whitelisted hosts, with or without the conventional www prefix, cannot be reported.
     */
    private function isWhitelistedReportHost(string $host): bool
    {
        $normalize = function (string $domain): ?string {
            $domain = $this->reportedLinkHost($domain);
            return $domain === null ? null : (string) preg_replace('/^www\./i', '', $domain);
        };
        $normalizedHost = $normalize($host);
        if ($normalizedHost === null)
            return false;

        $cutoff = time() - (WhitelistModel::PROTECTION_DAYS * 86400);
        foreach ((new WhitelistModel())->findAll() as $entry) {
            if ($normalize((string) ($entry['domain'] ?? '')) !== $normalizedHost)
                continue;
            $addedAt = strtotime((string) ($entry['created_at'] ?? ''));
            if ($addedAt !== false && $addedAt >= $cutoff)
                return true;
        }
        return false;
    }

    /**
     * Enforce rolling quotas per authenticated account and source IP.
     */
    private function reportRateLimitMessage(): ?string
    {
        $db = db_connect();
        $userId = (int) auth()->id();
        $ipAddress = $this->request->getIPAddress();
        $hourAgo = date('Y-m-d H:i:s', time() - 3600);
        $dayAgo = date('Y-m-d H:i:s', time() - 86400);
        $userHour = $db->table('audit_logs')->where('action', 'Signalement Utilisateur')->where('user_id', $userId)->where('created_at >=', $hourAgo)->countAllResults();
        $userDay = $db->table('audit_logs')->where('action', 'Signalement Utilisateur')->where('user_id', $userId)->where('created_at >=', $dayAgo)->countAllResults();
        $ipHour = $db->table('audit_logs')->where('action', 'Signalement Utilisateur')->where('ip_address', $ipAddress)->where('created_at >=', $hourAgo)->countAllResults();
        $ipDay = $db->table('audit_logs')->where('action', 'Signalement Utilisateur')->where('ip_address', $ipAddress)->where('created_at >=', $dayAgo)->countAllResults();

        if ($userHour >= self::USER_REPORT_LIMIT_PER_HOUR || $ipHour >= self::IP_REPORT_LIMIT_PER_HOUR) {
            return 'Limite atteinte : réessayez dans une heure.';
        }
        if ($userDay >= self::USER_REPORT_LIMIT_PER_DAY || $ipDay >= self::IP_REPORT_LIMIT_PER_DAY) {
            return 'Limite quotidienne de signalements atteinte.';
        }
        return null;
    }

    /**
     * Find active cards whose parsed hostname exactly matches the reported domain.
     */
    private function activeItemsForReportHost(string $host): array
    {
        $candidates = db_connect()
            ->table('item')
            ->select('id, titre, lien, link_status')
            ->where('deleted_at', null)
            ->where('id_division !=', ItemModel::PERSONAL_DIVISION_ID)
            ->where('lien IS NOT NULL', null, false)
            ->like('lien', $host)
            ->get()
            ->getResult();

        $matches = [];
        foreach ($candidates as $candidate) {
            if ($this->reportedLinkHost((string) $candidate->lien) === $host)
                $matches[] = $candidate;
        }
        return $matches;
    }

    /**
     * Record user reports without making unreliable network checks.
     */
    public function reportLink()
    {
        if (!auth()->loggedIn()) {
            return $this->response->setStatusCode(401)->setJSON([
                'success' => false,
                'message' => 'Vous devez être connecté pour signaler un lien.',
                'csrf_token' => csrf_hash(),
            ]);
        }

        $json = $this->request->getJSON();
        $itemId = filter_var($json->id ?? null, FILTER_VALIDATE_INT, ['options' => ['min_range' => 1]]);
        if ($itemId === false) {
            return $this->response->setStatusCode(400)->setJSON([
                'success' => false,
                'message' => 'ID de carte invalide.',
                'csrf_token' => csrf_hash(),
            ]);
        }

        $item = $this->model->find($itemId);
        if (!$item || empty($item->lien)) {
            return $this->response->setStatusCode(404)->setJSON([
                'success' => false,
                'message' => 'Carte introuvable ou lien vide.',
                'csrf_token' => csrf_hash(),
            ]);
        }
        if ((int) $item->id_division === ItemModel::PERSONAL_DIVISION_ID) {
            return $this->response->setStatusCode(403)->setJSON([
                'success' => false,
                'message' => 'Les cartes personnelles sont privées et ne peuvent pas être signalées.',
                'csrf_token' => csrf_hash(),
            ]);
        }

        $host = $this->reportedLinkHost((string) $item->lien);
        if ($host === null) {
            return $this->response->setStatusCode(400)->setJSON([
                'success' => false,
                'message' => 'Le lien de cette carte ne contient pas de domaine valide.',
                'csrf_token' => csrf_hash(),
            ]);
        }
        if ($rateLimitMessage = $this->reportRateLimitMessage()) {
            return $this->response->setStatusCode(429)->setJSON([
                'success' => false,
                'message' => $rateLimitMessage,
                'csrf_token' => csrf_hash(),
            ]);
        }
        if ($this->isWhitelistedReportHost($host)) {
            return $this->response->setJSON([
                'success' => false,
                'message' => 'Ce domaine a été confirmé fonctionnel récemment. Il pourra être signalé à nouveau après 7 jours.',
                'csrf_token' => csrf_hash(),
            ]);
        }

        $db = db_connect();
        $lockName = 'deadlink_' . substr(hash('sha256', $host), 0, 48);
        $lockedHost = $host;
        $lock = $db->query('SELECT GET_LOCK(?, 8) AS acquired', [$lockName])->getRowArray();
        if ((int) ($lock['acquired'] ?? 0) !== 1) {
            return $this->response->setStatusCode(503)->setJSON([
                'success' => false,
                'message' => 'Le signalement est temporairement occupé. Réessayez.',
                'csrf_token' => csrf_hash(),
            ]);
        }

        try {
            $db->transBegin();
            $item = $this->model->find($itemId);
            if (!$item || empty($item->lien)) {
                $db->transRollback();
                return $this->response->setStatusCode(404)->setJSON([
                    'success' => false,
                    'message' => 'Carte introuvable ou lien vide.',
                    'csrf_token' => csrf_hash(),
                ]);
            }
            if ((int) $item->id_division === ItemModel::PERSONAL_DIVISION_ID) {
                $db->transRollback();
                return $this->response->setStatusCode(403)->setJSON([
                    'success' => false,
                    'message' => 'Les cartes personnelles sont privées et ne peuvent pas être signalées.',
                    'csrf_token' => csrf_hash(),
                ]);
            }
            $host = $this->reportedLinkHost((string) $item->lien);
            if ($host === null || $host !== $lockedHost) {
                $db->transRollback();
                return $this->response->setJSON([
                    'success' => false,
                    'message' => 'Le lien de cette carte a changé. Rechargez la page avant de le signaler.',
                    'csrf_token' => csrf_hash(),
                ]);
            }
            if ($this->isWhitelistedReportHost($host)) {
                $db->transRollback();
                return $this->response->setJSON([
                    'success' => false,
                    'message' => 'Ce domaine a été confirmé fonctionnel récemment. Il pourra être signalé à nouveau après 7 jours.',
                    'csrf_token' => csrf_hash(),
                ]);
            }
            if ($this->domainAlreadyReported($host)) {
                $db->transRollback();
                return $this->response->setJSON([
                    'success' => false,
                    'message' => 'Ce domaine a déjà un signalement en attente ou traité.',
                    'csrf_token' => csrf_hash(),
                ]);
            }

            $domainItems = $this->activeItemsForReportHost($host);
            if ($domainItems === []) {
                throw new \RuntimeException('Aucune carte active ne correspond à ce domaine.');
            }
            foreach ($domainItems as $domainItem) {
                if (!$this->model->update($domainItem->id, ['link_status' => 'reported'])) {
                    throw new \RuntimeException('Impossible de mettre à jour le statut des cartes du domaine.');
                }
            }

            $now = date('Y-m-d H:i:s');
            $cronLogId = (new CronLogModel())->insert([
                'task_name' => CronLogModel::TASK_PENDING_USER_REPORT,
                'last_run' => $now,
                'item_id' => $itemId,
                'titre' => $item->titre,
                'url_testee' => 'https://' . $host . '/',
                'code_erreur' => 0,
            ]);
            if (!$cronLogId) {
                throw new \RuntimeException('Impossible d’enregistrer l’historique du signalement.');
            }

            $auditId = (new AuditLogModel())->logAction(
                'Signalement Utilisateur',
                'L’utilisateur ' . auth()->user()->username . " a signalé manuellement le domaine {$host} comme mort (carte ID {$itemId})."
            );
            if (!$auditId || !$db->transStatus()) {
                throw new \RuntimeException('Impossible d’enregistrer le journal d’audit.');
            }

            $db->transCommit();
            return $this->response->setJSON([
                'success' => true,
                'message' => 'Signalement enregistré. Merci pour votre aide !',
                'csrf_token' => csrf_hash(),
            ]);
        } catch (\Throwable $e) {
            $db->transRollback();
            log_message('error', '[ReportLink] Échec du signalement du domaine {host}: {message}', [
                'host' => $host,
                'message' => $e->getMessage(),
            ]);
            return $this->response->setStatusCode(500)->setJSON([
                'success' => false,
                'message' => 'Le signalement n’a pas pu être enregistré. Réessayez plus tard.',
                'csrf_token' => csrf_hash(),
            ]);
        } finally {
            $db->query('SELECT RELEASE_LOCK(?)', [$lockName]);
        }
    }
}
```

```php
        // de casser le fonctionnement si le .env ne contient pas encore TMDB_API_KEY.
        $this->tmdbApiKey = trim((string) (env('TMDB_API_KEY') ?: 'ba55da0439797150ed58c4e524584823'));
    }

    public function search()
    {
        $rawQuery = $this->request->getGet('q');
        $query = is_string($rawQuery) ? trim($rawQuery) : '';
        $rawType = $this->request->getGet('type');
        $type = is_string($rawType) ? strtolower(trim($rawType)) : '';

        if ('' === $query) {
            return $this->response->setJSON(['unified' => []]);
        }

        if (mb_strlen($query) > 200) {
            return $this->response->setStatusCode(400)->setJSON(['error' => 'Requête trop longue.']);
        }

        // Une URL reste traitée comme avant : récupération des métadonnées OpenGraph.
        if (filter_var($query, FILTER_VALIDATE_URL)) {
            if (!ExternalUrlGuard::isPublicHttpUrl($query)) {
                return $this->response->setStatusCode(400)->setJSON(['error' => 'URL invalide.']);
            }

            $metaData = $this->scrapeOpenGraph($query);

            return $this->response->setJSON($metaData ? [$metaData] : ['error' => 'Impossible de lire le lien.']);
        }

        $type = $this->normalizeType($type);
        $results = [];

        try {
            switch ($type) {
                case 'film':
                    $results = $this->searchTmdbMovies($query);

                    break;

                case 'serie':
                    $results = $this->searchTmdbSeries($query);

                    break;

                case 'anime':
                    $results = array_merge(
                        $this->searchTmdbAnime($query),
                        $this->searchAniList($query, 'ANIME'),
                        $this->searchMangaDex($query, true)
                    );

                    break;

                case 'manga':
                    $results = array_merge(
                        $this->searchAniList($query, 'MANGA'),
                        $this->searchMangaDex($query, false)
                    );

                    break;

                case 'video':
                    $results = $this->searchYouTube($query);

                    break;

                case 'streaming':
                    $results = $this->searchStreaming($query);

                    break;

                default:
                    // Pour une catégorie générique, on cherche dans les films/séries/anime/manga.
                    $results = array_merge(
                        $this->searchTmdbMovies($query, 4),
                        $this->searchTmdbSeries($query, 4),
                        $this->searchAniList($query, 'ANIME', 4),
                        $this->searchAniList($query, 'MANGA', 4)
                    );

                    break;
            }
        } catch (\Throwable $e) {
            log_message('error', 'MediaSearchController: '.$e->getMessage());
        }

        $results = $this->prepareResults($results, $query, $type);

        return $this->response->setJSON([
            'unified' => $results,
            'type' => $type,
            'count' => count($results),
        ]);
    }

    private function normalizeType(string $type): string
    {
        $type = str_replace(['é', 'è', 'ê', 'à'], ['e', 'e', 'e', 'a'], $type);

        if (str_contains($type, 'anime')) {
            return 'anime';
        }
        if (str_contains($type, 'manga')) {
            return 'manga';
        }
        if (str_contains($type, 'serie') || str_contains($type, 'tv')) {
            return 'serie';
        }
        if (str_contains($type, 'film') || str_contains($type, 'movie')) {
            return 'film';
        }
        if (str_contains($type, 'video') || str_contains($type, 'youtube')) {
            return 'video';
        }
        if (str_contains($type, 'stream')) {
            return 'streaming';
        }

        return 'autre';
    }

    private function searchAniList(string $query, string $mediaType, int $limit = 8): array
    {
        $graphql = <<<'GRAPHQL'
            query ($search: String!, $type: MediaType!, $perPage: Int) {
              Page(page: 1, perPage: $perPage) {
                media(search: $search, type: $type, isAdult: false) {
                  id
                  title { romaji english native }
                  description(asHtml: false)
                  startDate { year }
                  episodes
                  chapters
                  volumes
                  coverImage { extraLarge large medium }
                  format
                  countryOfOrigin
                }
              }
            }
            GRAPHQL;

        try {
            $response = $this->client->post('https://graphql.anilist.co', [
                'headers' => [
                    'Content-Type' => 'application/json',
                    'Accept' => 'application/json',
                ],
                'json' => [
                    'query' => $graphql,
                    'variables' => [
                        'search' => $query,
                        'type' => $mediaType,
                        'perPage' => $limit,
                    ],
                ],
            ]);

            if (200 !== $response->getStatusCode()) {
                return [];
            }

            $body = json_decode((string) $response->getBody(), true);
            $media = $body['data']['Page']['media'] ?? [];
            $results = [];

            foreach ($media as $item) {
                $title = (string) (
                    $item['title']['english']
                    ?? $item['title']['romaji']
                    ?? $item['title']['native']
                    ?? 'Inconnu'
                );

                $image = (string) (
                    $item['coverImage']['extraLarge']
                    ?? $item['coverImage']['large']
                    ?? $item['coverImage']['medium']
                    ?? ''
                );

                if ('' === $image) {
                    continue;
                }

                $year = (string) ($item['startDate']['year'] ?? '');
                $label = 'ANIME' === $mediaType ? 'ANIME' : 'MANGA';

                $results[] = [
                    'titre' => $title,
                    'imageThumb' => $image,
                    'imageLarge' => $image,
                    'description' => trim(strip_tags((string) ($item['description'] ?? ''))),
                    'info' => ($year ? $year.' - ' : '').$label,
                    'lien' => 'https://anilist.co/'.strtolower('ANIME' === $mediaType ? 'anime' : 'manga').'/'.(int) ($item['id'] ?? 0),
                    'total_episodes' => 'ANIME' === $mediaType ? (int) ($item['episodes'] ?? 0) : (int) ($item['chapters'] ?? 0),
                    'total_saisons' => 'MANGA' === $mediaType ? (int) ($item['volumes'] ?? 0) : '',
                    'seasons_data' => null,
                    'source' => 'AniList',
                    'sourcePriority' => 'MANGA' === $mediaType ? 110 : 95,
                    'externalId' => $item['id'] ?? null,
                    'mediaType' => strtolower($mediaType),
                    'providers' => [],
                    'score' => 0,
                ];
            }

            return $results;
        } catch (\Throwable $e) {
            log_message('warning', 'AniList indisponible: '.$e->getMessage());

            return [];
        }
    }

                    'sourcePriority' => 100,

    private function searchStreaming(string $query, int $limit = 8): array
    {
        // Les fournisseurs sont interrogés par titre : limiter les candidats
        // évite une rafale de requêtes séquentielles pour une seule recherche.
        $movies = $this->searchTmdbMovies($query, 2);
        $series = $this->searchTmdbSeries($query, 2, 0);
        $results = array_merge($movies, $series);

        foreach ($results as &$result) {
            $id = (int) ($result['externalId'] ?? 0);
            $mediaType = $result['mediaType'] ?? '';
            if ($id <= 0 || !in_array($mediaType, ['movie', 'tv'], true)) {
                continue;
            }

            $path = 'movie' === $mediaType
                ? '/movie/'.$id.'/watch/providers'
                : '/tv/'.$id.'/watch/providers';

            $providersBody = $this->tmdbGet($path, []);
            $fr = $providersBody['results']['FR'] ?? [];
            $providers = [];

            foreach (['flatrate', 'free', 'ads', 'rent', 'buy'] as $group) {
                foreach ($fr[$group] ?? [] as $provider) {
                    $name = trim((string) ($provider['provider_name'] ?? ''));
                    if ('' !== $name && !in_array($name, $providers, true)) {
                        $providers[] = $name;
                    }
                }
            }

            $result['providers'] = $providers;
            $result['lien'] = (string) ($fr['link'] ?? '');
            if ($providers) {
                $result['info'] .= ' • '.implode(', ', array_slice($providers, 0, 4));
            }
        }
        unset($result);

        return array_values(array_filter($results, static fn ($r) => !empty($r['providers'])));
    }

    private function prepareResults(array $results, string $query, string $type): array
    {
        foreach ($results as &$result) {
            $result['score'] = $this->scoreTitle($query, (string) ($result['titre'] ?? ''));

            // Bonus pour une source adaptée à la catégorie demandée.
            $result['score'] += match ($type) {
                'film', 'serie' => (($result['source'] ?? '') === 'TMDB' ? 10 : 0),
                'anime' => (($result['source'] ?? '') === 'TMDB' ? 12 : (($result['source'] ?? '') === 'AniList' ? 10 : 0)),
                'manga' => (($result['source'] ?? '') === 'AniList' ? 12 : (($result['source'] ?? '') === 'MangaDex' ? 9 : 0)),
                'video' => (($result['source'] ?? '') === 'YouTube' ? 10 : 0),
                'streaming' => (($result['source'] ?? '') === 'TMDB' ? 10 : 0),
                default => 0,
            };

            if (!empty($result['imageLarge'])) {
                $result['score'] += 5;
            }

            $result['score'] = max(0, min(100, (int) round($result['score'])));
        }
        unset($result);

        // Déduplication : le même titre provenant de plusieurs APIs devient une seule carte.
        $deduped = [];
        foreach ($results as $result) {
            if (empty($result['titre'])) {
                continue;
            }

            // On retire les titres très courts qui donnent beaucoup de faux positifs.
            if (mb_strlen(trim((string) $result['titre'])) < 2) {
                continue;
            }

            $key = $this->normalizeTitle((string) $result['titre']);
            $year = preg_match('/\b(19|20)\d{2}\b/', (string) ($result['info'] ?? ''), $matches) ? $matches[0] : '';
            $dedupeKey = $key.'|'.$year;

            if (!isset($deduped[$dedupeKey])) {
                $result['sources'] = [$result['source'] ?? 'API'];
                $deduped[$dedupeKey] = $result;

                continue;
            }

            $existing = $deduped[$dedupeKey];
            $existingSources = $existing['sources'] ?? [$existing['source'] ?? 'API'];
            $newSource = $result['source'] ?? 'API';
            if (!in_array($newSource, $existingSources, true)) {
                $existingSources[] = $newSource;
            }

            // On garde la meilleure image selon la priorité de source, pas simplement le dernier résultat.
            if (($result['sourcePriority'] ?? 0) > ($existing['sourcePriority'] ?? 0)) {
                $result['sources'] = $existingSources;
                if (empty($result['description']) && !empty($existing['description'])) {
                    $result['description'] = $existing['description'];
                }
                if (empty($result['seasons_data']) && !empty($existing['seasons_data'])) {
                    $result['seasons_data'] = $existing['seasons_data'];
                }
                $deduped[$dedupeKey] = $result;
            } else {
                if (empty($existing['description']) && !empty($result['description'])) {
                    $existing['description'] = $result['description'];
                }
                if (empty($existing['seasons_data']) && !empty($result['seasons_data'])) {
                    $existing['seasons_data'] = $result['seasons_data'];
                }
                if (empty($existing['lien']) && !empty($result['lien'])) {
                    $existing['lien'] = $result['lien'];
                }
                $existing['score'] = max((int) $existing['score'], (int) $result['score']);
                $existing['sources'] = $existingSources;
                $deduped[$dedupeKey] = $existing;
            }
        }

        $results = array_values($deduped);

        usort($results, static function (array $a, array $b): int {
            $scoreCompare = ($b['score'] ?? 0) <=> ($a['score'] ?? 0);
            if (0 !== $scoreCompare) {
                return $scoreCompare;
            }

            return ($b['sourcePriority'] ?? 0) <=> ($a['sourcePriority'] ?? 0);
        });

        // Pas de résultat faible : on préfère peu de bons résultats à une liste de faux positifs.
        $results = array_values(array_filter($results, static fn (array $r): bool => ($r['score'] ?? 0) >= 42));

        foreach ($results as &$result) {
            $result['sourcesLabel'] = implode(' + ', $result['sources'] ?? [$result['source'] ?? 'API']);
            unset($result['sourcePriority'], $result['externalId'], $result['sources']);
        }
        unset($result);

        return array_slice($results, 0, 8);
    }

    private function scoreTitle(string $query, string $title): int
    {
        $q = $this->normalizeTitle($query);
        $t = $this->normalizeTitle($title);

        if ('' === $q || '' === $t) {
            return 0;
        }
        if ($q === $t) {
            return 90;
        }
        if (str_contains($t, $q)) {
            return 78;
        }
        if (str_contains($q, $t)) {
            return 72;
        }

        similar_text($q, $t, $percent);
        $score = (int) round($percent * 0.72);

        $qTokens = array_values(array_filter(explode(' ', $q), static fn ($v) => mb_strlen($v) >= 2));
        $tTokens = array_values(array_filter(explode(' ', $t), static fn ($v) => mb_strlen($v) >= 2));
        if ($qTokens && $tTokens) {
            $intersection = count(array_intersect($qTokens, $tTokens));
            $score += (int) round(($intersection / count($qTokens)) * 25);
        }

        return min(90, $score);
    }

    private function normalizeTitle(string $value): string
    {
        $value = trim(mb_strtolower($value));
        $converted = @iconv('UTF-8', 'ASCII//TRANSLIT//IGNORE', $value);
        if (false !== $converted) {
            $value = $converted;
        }
        $value = preg_replace('/[^a-z0-9]+/i', ' ', $value) ?? $value;

        return trim(preg_replace('/\s+/', ' ', $value) ?? $value);
    }

    private function tmdbGet(string $path, array $query = []): array
    {
        if ('' === $this->tmdbApiKey) {
            return [];
        }

        $query['api_key'] = $this->tmdbApiKey;
        $query['language'] ??= 'fr-FR';

        try {
            $response = $this->client->get('https://api.themoviedb.org/3'.$path, ['query' => $query]);
            if (200 !== $response->getStatusCode()) {
                return [];
            }

            return json_decode((string) $response->getBody(), true) ?: [];
        } catch (\Throwable $e) {
            log_message('warning', 'TMDB indisponible: '.$e->getMessage());

            return [];
        }
    }

    private function scrapeOpenGraph(string $url): ?array
    {
        if (!ExternalUrlGuard::isPublicHttpUrl($url)) {
            return null;
        }

        try {
            $response = $this->client->get($url, [
                'headers' => ['Accept-Language' => 'fr-FR,fr;q=0.9,en;q=0.8'],
                'allow_redirects' => false,
            ]);
            if ($response->getStatusCode() >= 400) {
                return null;
            }

            $html = (string) $response->getBody();
            if ('' === $html) {
                return null;
            }

            $doc = new \DOMDocument();
            @$doc->loadHTML(mb_convert_encoding($html, 'HTML-ENTITIES', 'UTF-8'));
            $tags = $doc->getElementsByTagName('meta');

            $data = [
                'titre' => '',
                'description' => '',
                'image' => '',
                'lien' => $url,
                'is_link' => true,
            ];

            foreach ($tags as $tag) {
                $property = $tag->getAttribute('property');
                $name = $tag->getAttribute('name');
                $content = $tag->getAttribute('content');

                if ('og:title' === $property || 'twitter:title' === $name) {
                    $data['titre'] = $content;
                } elseif ('og:description' === $property || 'description' === $name) {
                    $data['description'] = $content;
                } elseif ('og:image' === $property || 'twitter:image' === $name) {
                    $data['image'] = $content;
                }
            }

            if ('' === $data['titre']) {
                $titles = $doc->getElementsByTagName('title');
                if ($titles->length > 0) {
                    $data['titre'] = trim((string) $titles->item(0)->nodeValue);
                }
            }

            return $data;
        } catch (\Throwable $e) {
            return null;
        }
    }
}
```

```php
    /**
     * Traite la suppression définitive du compte utilisateur (Conformité RGPD)
     * et conserve les contributions publiques.
     */
    public function deleteAccount()
    {
        $user = auth()->user();
        $userId = (int) $user->id;

        $rules = [
            'password_for_deletion' => 'required',
        ];

        if (!$this->validate($rules)) {
            return redirect()->to('p?tab=danger')->with('error', 'Le mot de passe est requis pour supprimer le compte.');
        }

        $credentials = [
            'email'    => $user->email,
            'password' => $this->request->getPost('password_for_deletion'),
        ];
        $authenticator = auth('session')->getAuthenticator();
        $result = $authenticator->check($credentials);

        if (!$result->isOK()) {
            return redirect()->to('p?tab=danger')->with('error', 'Mot de passe incorrect. Suppression annulée.');
        }

        $itemModel = new ItemModel();

        // 1. Sauvegarde des contributions : Transfert des cartes publiques (is_public = 1) au Super Admin (ID = 1)
        $itemModel->builder()
            ->where('id_user', $userId)
            ->where('is_public', 1)
            ->update(['id_user' => 1]);

        // 2. Suppression forte des cartes restantes (privées ou en attente d'approbation)
        $itemModel->builder()
            ->where('id_user', $userId)
            ->delete();

        // 3. Suppression des propositions de modifications (brouillons) en attente
        $revisionModel = new ItemRevisionModel();
        $revisionModel->where('id_user', $userId)->delete();

        $audit = new AuditLogModel();
        $audit->logAction('Suppression Compte', "L'utilisateur ID {$userId} ({$user->username}) a supprimé son compte définitivement. Ses cartes publiques ont été transférées à l'administration.");

        // 4. Suppression de l'utilisateur
        $users = auth()->getProvider();
        $users->delete($userId, true);

        // 5. Déconnexion
        auth()->logout();

        return redirect()->to('/')->with('message', 'Votre compte a été définitivement supprimé. Vos cartes publiques ont été conservées et anonymisées.');
    }

    /**
     * Formate la date d'inscription en français (ex: "septembre 2026").
     */
    private function formatMemberSince(mixed $createdAt): string
    {
        if (empty($createdAt)) {
            return 'Membre vérifié';
        }

        $timestamp = is_object($createdAt) && method_exists($createdAt, 'getTimestamp')
            ? $createdAt->getTimestamp()
            : strtotime((string) $createdAt);

        if (false === $timestamp) {
            return 'Membre vérifié';
        }

        $months = [
            1 => 'janvier',
            2 => 'février',
            3 => 'mars',
            4 => 'avril',
            5 => 'mai',
            6 => 'juin',
            7 => 'juillet',
            8 => 'août',
            9 => 'septembre',
            10 => 'octobre',
            11 => 'novembre',
            12 => 'décembre',
        ];

        $monthName = $months[(int) date('n', $timestamp)] ?? '';

        return 'Depuis ' . $monthName . ' ' . date('Y', $timestamp);
    }
```

```php
        );
```

## app/Filters/HtmlMinifier.php
```php
<?php

declare(strict_types=1);

namespace App\Filters;

use CodeIgniter\Filters\FilterInterface;
use CodeIgniter\HTTP\RequestInterface;
use CodeIgniter\HTTP\ResponseInterface;

class HtmlMinifier implements FilterInterface
{
    public function before(RequestInterface $request, $arguments = null): void
    {
        // Rien à faire avant l'exécution du contrôleur
    }

    public function after(RequestInterface $request, ResponseInterface $response, $arguments = null): void
    {
        $contentType = $response->getHeaderLine('Content-Type');
        if (empty($contentType) && method_exists($response, 'getContentType')) {
            $contentType = $response->getContentType();
        }

        if (str_contains($contentType ?: 'text/html', 'text/html')) {
            $html = (string) $response->getBody();

            // Protège les blocs sensibles aux retours à la ligne et espaces
            $preserved = [];
            $html = preg_replace_callback(
                '#<(script|style|textarea|pre)\b[^>]*>.*?</\1>#is',
                static function (array $matches) use (&$preserved): string {
                    $key = '___MINIFY_PRESERVE_' . count($preserved) . '___';
                    $preserved[$key] = $matches[0];

                    return $key;
                },
                $html
            ) ?? $html;

            $search = [
                '/\>[^\S ]+/s',
                '/[^\S ]+\</s',
                '/(\s)+/s',
                '/<!--(.*?)-->/s',
            ];

            $replace = [
                '>',
                '<',
                '\1',
                '',
            ];

            $minifiedHtml = preg_replace($search, $replace, $html) ?? $html;

            if ($preserved !== []) {
                $minifiedHtml = strtr($minifiedHtml, $preserved);
            }

            $response->setBody($minifiedHtml);
        }
    }
}
```

## app/Libraries/ExternalUrlGuard.php
```php
<?php

declare(strict_types=1);

namespace App\Libraries;

final class ExternalUrlGuard
{
    public static function isPublicHttpUrl(string $url): bool
    {
        if ('' === $url || strlen($url) > 2048 || !filter_var($url, FILTER_VALIDATE_URL)) {
            return false;
        }

        $parts = parse_url($url);
        if (!is_array($parts)) {
            return false;
        }

        $scheme = strtolower((string) ($parts['scheme'] ?? ''));
        $host = strtolower(rtrim((string) ($parts['host'] ?? ''), '.'));
        if (!in_array($scheme, ['http', 'https'], true) || '' === $host || isset($parts['user']) || isset($parts['pass'])) {
            return false;
        }

        $ipHost = trim($host, '[]');
        if (false !== filter_var($ipHost, FILTER_VALIDATE_IP)) {
            return false !== filter_var(
                $ipHost,
                FILTER_VALIDATE_IP,
                FILTER_FLAG_NO_PRIV_RANGE | FILTER_FLAG_NO_RES_RANGE,
            );
        }

        if (!str_contains($host, '.')) {
            return false;
        }

        return !preg_match('/\.(?:localhost|local|localdomain|lan|internal|intranet|test)$/i', $host);
    }
}
```

```php
     * Le découpage En cours / En pause est ensuite effectué

    {
        $rows = $this->db
            ->table($this->table)
            ->select(
                'titre,
                status,
                episode_global,
                total_episodes_global,
                episode,
                total_episodes,
                saison,
                total_saisons'
            )
            ->where('id_user', $userId)
            ->whereIn('status', ['En cours', 'En pause'])
            ->whereIn('id_division', [1, 4])
            ->where('deleted_at IS NULL', null, false)
            ->orderBy('position', 'ASC')
            ->get()
            ->getResult()
        ;

        $series = [
            'en_cours' => [],
            'en_pause' => [],
        ];

        foreach ($rows as $row) {
            // Calcul identique à l'ancien fonctionnement.
            $row->vu_global = null !== $row->episode_global
                ? max(0, (int) $row->episode_global - 1)
                : null;

            if (
                null !== $row->total_episodes_global
                || null !== $row->total_episodes
            ) {
                $totalEpisodes = $row->total_episodes_global
                    ?? $row->total_episodes;

                $episodeActuel = $row->episode_global
                    ?? $row->episode
                    ?? 1;

                $row->reste_global = max(
                    0,
                    (int) $totalEpisodes - (int) $episodeActuel + 1
                );
            } else {
                $row->reste_global = null;
            }

            if (
                null !== $row->total_saisons
                && null !== $row->saison
            ) {
                $row->reste_s = max(
                    0,
                    (int) $row->total_saisons - (int) $row->saison + 1
                );
            } else {
                $row->reste_s = null;
            }

            if ('En cours' === $row->status) {
                $series['en_cours'][] = $row;
            } else {
                $series['en_pause'][] = $row;
            }
        }

        return $series;
    }
```

## app/Models/WhitelistModel.php
```php
<?php

declare(strict_types=1);

namespace App\Models;

use CodeIgniter\Model;

class WhitelistModel extends Model
{
    public const PROTECTION_DAYS = 7;

    protected $table            = 'whitelist';
    protected $primaryKey       = 'id';
    protected $useAutoIncrement = true;
    protected $returnType       = 'array';
    protected $useSoftDeletes   = false;
    protected $allowedFields    = ['domain', 'created_at'];
    protected $useTimestamps    = false;

    /**
     * Retourne tous les domaines déclarés comme valides, du plus récent au plus ancien.
     */
    public function getAllOrdered(): array
    {
        return $this->orderBy('created_at', 'DESC')->findAll();
    }

    /**
     * Ajoute ou renouvelle la date de validation d'un domaine dans la whitelist.
     */
    public function upsertDomain(string $host): bool
    {
        $now      = date('Y-m-d H:i:s');
        $existing = $this->where('domain', $host)->first();

        if ($existing) {
            return (bool) $this->update($existing['id'], ['created_at' => $now]);
        }

        return (bool) $this->insert([
            'domain'     => $host,
            'created_at' => $now,
        ]);
    }

    /**
     * Indique si le délai de protection de 7 jours est toujours actif.
     */
    public function isProtectionActive(?string $createdAt): bool
    {
        if (empty($createdAt)) {
            return false;
        }

        $timestamp = strtotime($createdAt);
        if ($timestamp === false) {
            return false;
        }

        return ($timestamp + (self::PROTECTION_DAYS * 86400)) > time();
    }

    /**
     * Calcule la date de fin de protection (created_at + 7 jours).
     */
    public function getProtectionExpiresAt(?string $createdAt): ?string
    {
        if (empty($createdAt)) {
            return null;
        }

        $timestamp = strtotime($createdAt);
        if ($timestamp === false) {
            return null;
        }

        return date('Y-m-d H:i:s', $timestamp + (self::PROTECTION_DAYS * 86400));
    }
}
```
