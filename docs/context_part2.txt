```javascript
(function () {
    'use strict';

    const CACHE_BASE_PREFIX = 'cat_cache_v1_';
    const AUTH_STATE_KEY = 'nav_auth_state_v1';
    const CACHE_TTL_MS = 2 * 60 * 1000; // 2 minutes
    const pendingFetches = new Map();

    function getAuthState() {
        return document.querySelector('.btn-logout') ? 'auth' : 'guest';
    }

    function getCachePrefix() {
        return `${CACHE_BASE_PREFIX}${getAuthState()}_`;
    }

    function clearAllNavCaches() {
        try {
            for (let i = sessionStorage.length - 1; i >= 0; i--) {
                const k = sessionStorage.key(i);
                if (k && (k.startsWith(CACHE_BASE_PREFIX) || k.startsWith('prof_cache_v1_'))) {
                    sessionStorage.removeItem(k);
                }
            }
        } catch (_) {}
    }

    function syncAuthCacheState() {
        const currentState = getAuthState();
        try {
            const previousState = sessionStorage.getItem(AUTH_STATE_KEY);
            if (previousState !== currentState) {
                clearAllNavCaches();
                sessionStorage.setItem(AUTH_STATE_KEY, currentState);
            }
        } catch (_) {}
    }

    function normalizeCategoryUrl(rawUrl) {
        const parsed = new URL(rawUrl, window.location.origin);
        parsed.search = '';
        parsed.hash = '';
        return parsed.toString();
    }

    function getCachedHtml(key) {
        try {
            const raw = sessionStorage.getItem(getCachePrefix() + key);
            if (!raw) return null;
            const entry = JSON.parse(raw);
            if (Date.now() - entry.ts > CACHE_TTL_MS) {
                sessionStorage.removeItem(getCachePrefix() + key);
                return null;
            }
            return entry.html;
        } catch (_) {
            return null;
        }
    }

    function setCachedHtml(key, html) {
        try {
            sessionStorage.setItem(getCachePrefix() + key, JSON.stringify({ html, ts: Date.now() }));
        } catch (_) {
            clearAllNavCaches();
        }
    }

    function removeCachedHtml(key) {
        try {
            sessionStorage.removeItem(getCachePrefix() + key);
        } catch (_) {}
    }

    function getBaseUrl() {
        const base = (window.siteConfig && window.siteConfig.baseUrl) ? window.siteConfig.baseUrl : '/';
        return base.endsWith('/') ? base : base + '/';
    }

    function updateNavActive(targetUrl) {
        const targetKey = normalizeCategoryUrl(targetUrl);
        document.querySelectorAll('nav.category-nav .nav-tab').forEach((tab) => {
            tab.classList.toggle('active', normalizeCategoryUrl(tab.href) === targetKey);
        });
    }

    async function fetchCategoryHtml(url) {
        const key = normalizeCategoryUrl(url);
        const cached = getCachedHtml(key);
        if (cached !== null) {
            return cached;
        }
        if (pendingFetches.has(key)) {
            return pendingFetches.get(key);
        }

        const promise = fetch(key, {
            headers: { 'X-Requested-With': 'XMLHttpRequest' },
            credentials: 'same-origin',
        })
            .then((res) => {
                if (!res.ok) throw new Error(`HTTP ${res.status}`);
                return res.text();
            })
            .then((html) => {
                const doc = new DOMParser().parseFromString(html, 'text/html');
                const main = doc.querySelector('main.container');
                if (!main) throw new Error('Contenu introuvable');
                const inner = main.innerHTML;
                setCachedHtml(key, inner);
                return inner;
            })
            .finally(() => {
                pendingFetches.delete(key);
            });

        pendingFetches.set(key, promise);
        return promise;
    }

    // --- Ré-initialisation des fonctionnalités après injection du HTML ---

    function initReleaseBadges(container) {
        let dismissed = [];
        try {
            dismissed = JSON.parse(localStorage.getItem('dismissedReleases') || '[]');
        } catch (_) {}

        container.querySelectorAll('.release-badge').forEach((badge) => {
            const key = badge.getAttribute('data-key');
            if (!dismissed.includes(key)) {
                badge.style.display = 'inline-block';
                badge.classList.add('fade-in');
                badge.addEventListener('click', function () {
                    this.style.display = 'none';
                    dismissed.push(key);
                    try {
                        localStorage.setItem('dismissedReleases', JSON.stringify(dismissed));
                    } catch (_) {}
                });
            }
        });
    }

    function initSearchAndFilters(container) {
        const searchInput = container.querySelector('#liveSearch');
        if (!searchInput) return;

        const cards = Array.from(container.querySelectorAll('.searchable-card'));
        const statusFilter = container.querySelector('#statusFilter');
        const categoryFilter = container.querySelector('#categoryFilter');
        const resultCount = container.querySelector('#cardResultCount');
        const filterKey = 'cardFilters';

        const categories = [...new Set(cards.map((c) => c.dataset.category).filter(Boolean))].sort((a, b) => a.localeCompare(b, 'fr'));
        categories.forEach((category) => {
            const option = document.createElement('option');
            option.value = category;
            option.textContent = category;
            categoryFilter?.appendChild(option);
        });

        const statuses = [...new Set(cards.map((c) => c.dataset.status).filter(Boolean))].sort((a, b) => a.localeCompare(b, 'fr'));
        statuses.forEach((value) => {
            const option = document.createElement('option');
            option.value = value;
            option.textContent = value;
            statusFilter?.appendChild(option);
        });

        try {
            const saved = JSON.parse(localStorage.getItem(filterKey) || '{}');
            searchInput.value = saved.query || '';
            if (statusFilter) statusFilter.value = saved.status || '';
            if (categoryFilter) categoryFilter.value = saved.category || '';

            if (categoryFilter && !Array.from(categoryFilter.options).some((o) => o.value === categoryFilter.value)) {
                categoryFilter.value = '';
            }
            if (statusFilter && !Array.from(statusFilter.options).some((o) => o.value === statusFilter.value)) {
                statusFilter.value = '';
            }
        } catch (_) {}

        const applyFilters = () => {
            const query = searchInput.value.trim().toLocaleLowerCase('fr');
            const status = statusFilter?.value || '';
            const category = categoryFilter?.value || '';
            let visible = 0;

            cards.forEach((card) => {
                const title = card.querySelector('.card-title, .search-target-title')?.innerText.toLocaleLowerCase('fr') || '';
                const desc = card.querySelector('.card-desc, .search-target-desc')?.innerText.toLocaleLowerCase('fr') || '';
                const match =
                    (!query || title.includes(query) || desc.includes(query)) &&
                    (!status || card.dataset.status === status) &&
                    (!category || card.dataset.category === category);
                card.style.display = match ? '' : 'none';
                if (match) visible++;
            });

            let openByDefault = false;
            try {
                openByDefault = localStorage.getItem('defaultOpenDivisions') === 'true';
            } catch (_) {}

            const hasFilter = Boolean(query || status || category);

            container.querySelectorAll('details.division-section, details.subcategory-details').forEach((detail) => {
                const hasVisible = Array.from(detail.querySelectorAll('.searchable-card')).some((c) => c.style.display !== 'none');
                detail.hidden = !hasVisible;
                if (hasFilter) {
                    detail.open = hasVisible;
                } else if (openByDefault && detail.classList.contains('division-section')) {
                    detail.open = hasVisible;
                }
            });

            if (resultCount) resultCount.textContent = `${visible} carte${visible > 1 ? 's' : ''}`;

            try {
                localStorage.setItem(filterKey, JSON.stringify({ query: searchInput.value, status, category }));
            } catch (_) {}
        };

        searchInput.addEventListener('input', applyFilters);
        statusFilter?.addEventListener('change', applyFilters);
        categoryFilter?.addEventListener('change', applyFilters);
        applyFilters();
        window.applyCardFilters = applyFilters;

    async function checkCardDispo(itemId, targetUrl) {
        const liveStatus = document.getElementById(`live-status-${itemId}`);
        if (!liveStatus || !targetUrl) return;

        const domains = Array.isArray(window.siteSupportedDomains) ? window.siteSupportedDomains : [];
        if (!domains.some((domain) => targetUrl.includes(domain))) {
            liveStatus.style.display = 'none';
            return;
        }

        const cacheKey = `dispo_check_${itemId}_${targetUrl}`;
        try {
            const cachedResult = sessionStorage.getItem(cacheKey);
            if (cachedResult) {
                const cacheData = JSON.parse(cachedResult);
                if ((Date.now() - cacheData.timestamp) / 60000 < 60) {
                    liveStatus.style.display = 'none';
                    return;
                }
            }
        } catch (_) {}

        liveStatus.style.color = 'var(--info)';
        liveStatus.textContent = 'Vérification...';

        try {
            const resp = await fetch(`${getBaseUrl()}q?urlCible=${encodeURIComponent(targetUrl)}`, {
                headers: { 'X-Requested-With': 'XMLHttpRequest' },
                credentials: 'same-origin',
            });

            if (!resp.ok) {
                liveStatus.style.display = 'none';
                return;
            }

            const data = await resp.json();
            if (data && data.success) {
                try {
                    sessionStorage.setItem(cacheKey, JSON.stringify({
                        timestamp: Date.now(),
                        disponible: Boolean(data.disponible),
                    }));
                } catch (_) {}

                if (data.disponible) {
                    liveStatus.style.color = 'var(--success)';
                    liveStatus.textContent = '✔ Disponible';
                } else {
                    liveStatus.style.display = 'none';
                }
            } else {
                liveStatus.style.display = 'none';
            }
        } catch (_) {
            liveStatus.style.display = 'none';
        }
    }

    async function switchCategory(url, pushHistory = true) {
        const mainEl = document.querySelector('main.container');
        if (!mainEl) {
            window.location.href = url;
            return;
        }

        const key = normalizeCategoryUrl(url);
        const token = ++activeNavToken;

        updateNavActive(key);
        if (pushHistory && normalizeCategoryUrl(window.location.href) !== key) {
            window.history.pushState({ categoryUrl: key }, '', key);
        }

        try {
            const html = await fetchCategoryHtml(key);
            if (token !== activeNavToken) return;

            mainEl.innerHTML = html;
            try {
                sessionStorage.setItem('last_cards_url', key);
            } catch (_) {}
            bindCategoryEvents(mainEl);
        } catch (_) {
            if (token === activeNavToken) {
                window.location.href = url;
            }
        }
    }

        // Synchronise le cache avec l'état de connexion actuel (vide automatiquement après login/logout)
        syncAuthCacheState();

        // Vide le cache dès le clic sur Déconnexion
        const logoutBtn = document.querySelector('.btn-logout');
        if (logoutBtn) {
            logoutBtn.addEventListener('click', clearAllNavCaches);
        }

        const navTabs = document.querySelectorAll('nav.category-nav .nav-tab');
        const mainEl = document.querySelector('main.container');

        if (!navTabs.length || !mainEl) return;

        if (new URL(window.location.href).searchParams.has('open')) {
            clearAllNavCaches();
        }

        const currentKey = normalizeCategoryUrl(window.location.href);
        setCachedHtml(currentKey, mainEl.innerHTML);

        navTabs.forEach((tab) => {
            tab.addEventListener('mouseenter', () => {
                fetchCategoryHtml(tab.href).catch(() => {});
            });
            tab.addEventListener('focus', () => {
                fetchCategoryHtml(tab.href).catch(() => {});
            });
            tab.addEventListener('click', (e) => {
                if (e.metaKey || e.ctrlKey || e.shiftKey || e.altKey || e.button !== 0) return;
                e.preventDefault();
                switchCategory(tab.href, true);
            });
        });

        const prefetchQueue = Array.from(navTabs)
            .map((tab) => normalizeCategoryUrl(tab.href))
            .filter((key) => getCachedHtml(key) === null);

        const prefetchNext = async () => {
            while (prefetchQueue.length > 0) {
                const nextKey = prefetchQueue.shift();
                if (!nextKey || getCachedHtml(nextKey) !== null) continue;
                try {
                    await fetchCategoryHtml(nextKey);
                } catch (_) {}
                await new Promise((r) => setTimeout(r, 150));
            }
        };

        if ('requestIdleCallback' in window) {
            window.requestIdleCallback(() => prefetchNext(), { timeout: 1500 });
        } else {
            setTimeout(prefetchNext, 500);
        }

        window.addEventListener('popstate', () => {
            if (document.querySelector('nav.category-nav .nav-tab')) {
                switchCategory(window.location.href, false);
            }
        });
    });
```

```javascript
        if (text.includes('stream')) return 'streaming';

    async function runSearch() {
        const searchId = ++activeSearchId;
        const button = document.getElementById('btn-api-search');
        const input = document.getElementById('titre');
        const container = document.getElementById('api-results-container');
        const status = document.getElementById('api-status');
        const clearButton = document.getElementById('clear-api-search');

        if (!button || !input || !container) return;

        const query = input.value.trim();
        if (!query) {
            if (typeof window.showToast === 'function') window.showToast("Entre d'abord un titre ou un lien !", 'danger');
            return;
        }

        let type = detectType();
        if (/^https?:\/\//i.test(query)) type = 'autre';

        if (status) {
            status.style.display = 'inline';
            status.textContent = 'Recherche multi-sources...';
        }
        button.disabled = true;
        container.innerHTML = '';
        container.style.display = 'none';

        const baseUrl = window.siteConfig.baseUrl.endsWith('/') ? window.siteConfig.baseUrl : window.siteConfig.baseUrl + '/';
        const url = `${baseUrl}f?q=${encodeURIComponent(query)}&type=${encodeURIComponent(type)}`;

        try {
            const response = await fetch(url, {
                headers: { 'X-Requested-With': 'XMLHttpRequest' }
            });

            if (!response.ok) throw new Error(`HTTP ${response.status}`);

            const data = await response.json();
            if (searchId !== activeSearchId) return;
            if (data.error) throw new Error(data.error);

            let results = Array.isArray(data.unified) ? data.unified : [];

            const description = document.getElementById('description');
            const maxLength = description && description.hasAttribute('maxlength')
                ? parseInt(description.getAttribute('maxlength'), 10)
                : 250;
            const limitCut = maxLength > 3 ? maxLength - 3 : maxLength;

            if (Array.isArray(data) && data.length && data[0].is_link) {
                results = [{
                    titre: data[0].titre,
                    imageThumb: data[0].image,
                    imageLarge: data[0].image,
                    description: data[0].description || '',
                    info: 'LIEN WEB',
                    lien: data[0].lien,
                    score: 100,
                    source: 'OpenGraph',
                    sourcesLabel: 'OpenGraph'
                }];
            }

            if (!results.length) {
                if (clearButton) clearButton.hidden = false;
                if (status) status.textContent = type === 'video' ? 'Aucune vidéo trouvée (vérifie YOUTUBE_API_KEY).' : 'Aucun résultat suffisamment pertinent';
                return;
            }

            if (status) status.textContent = `${results.length} résultat(s)`;
            container.style.display = 'block';
            if (clearButton) clearButton.hidden = false;

            results.forEach(result => container.appendChild(makeResult(result, limitCut)));
        } catch (error) {
            if (searchId !== activeSearchId) return;
            console.error('Auto-remplissage multi-sources :', error);
            if (status) status.textContent = 'Erreur de recherche';
            if (typeof window.showToast === 'function') window.showToast('Impossible de récupérer les résultats.', 'danger');
        } finally {
            if (searchId === activeSearchId) button.disabled = false;
        }
    }

    function init() {
        const button = document.getElementById('btn-api-search');
        const container = document.getElementById('api-results-container');
        if (!button || !container) return;
        const clearButton = document.getElementById('clear-api-search');
        clearButton?.addEventListener('click', () => {
            activeSearchId++;
            button.disabled = false;
            container.replaceChildren();
            container.style.display = 'none';
            const status = document.getElementById('api-status');
            if (status) { status.textContent = ''; status.style.display = 'none'; }
            clearButton.hidden = true;
        });

        // Capture avant le listener historique de script.js : cela empêche l'ancien
        // moteur TMDB/MangaDex de lancer une seconde recherche.
        button.addEventListener('click', function (event) {
            event.preventDefault();
            event.stopImmediatePropagation();
            runSearch();
        }, true);

        document.addEventListener('click', function (event) {
            if (!container.contains(event.target) && event.target !== button && event.target.id !== 'titre') {
                container.style.display = 'none';
            }
        });

        // Recherche automatique quand l'utilisateur arrête de taper un titre.
        const titleInput = document.getElementById('titre');
        let timer = null;
        if (titleInput) {
            titleInput.addEventListener('input', function () {
                if (suppressNextTitleSearch) { suppressNextTitleSearch = false; return; }
                clearTimeout(timer);
                const value = this.value.trim();
                if (value.length < 3 || /^https?:\/\//i.test(value)) return;
                timer = setTimeout(runSearch, 650);
            });
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', init, { once: true });
    } else {
        init();
    }
})();
```

```javascript
    async function fetchTabHtml(url) {
        const key = normalizeTabUrl(url);
        const cached = getCachedTab(key);
        if (cached !== null) {
            return cached;
        }
        if (pendingFetches.has(key)) {
            return pendingFetches.get(key);
        }

        const promise = fetch(key, {
            headers: { 'X-Requested-With': 'XMLHttpRequest' },
            credentials: 'same-origin',
        })
            .then((res) => {
                if (!res.ok) throw new Error(`HTTP ${res.status}`);
                return res.text();
            })
            .then((html) => {
                const doc = new DOMParser().parseFromString(html, 'text/html');
                const content = doc.querySelector('.profile-content');
                if (!content) throw new Error('Contenu profil introuvable');
                const inner = content.innerHTML;
                setCachedTab(key, inner);
                return inner;
            })
            .finally(() => {
                pendingFetches.delete(key);
            });

        pendingFetches.set(key, promise);
        return promise;
    }
```

```javascript
window.copierLien = function(lien) {
    function notifierSucces() {
        if (typeof showToast === 'function') {
            showToast('Lien copié dans le presse-papiers !', 'success');
        } else {
            alert('Lien copié !');
        }
    }

    function notifierErreur() {
        if (typeof showToast === 'function') {
            showToast('Erreur lors de la copie du lien.', 'danger');
        } else {
            alert('Erreur lors de la copie.');
        }
    }

    if (navigator.clipboard && window.isSecureContext) {
        navigator.clipboard.writeText(lien).then(() => {
            notifierSucces();
        }).catch(err => {
            console.error('Erreur API Clipboard :', err);
            notifierErreur();
        });
    } else {
        let textArea = document.createElement("textarea");
        textArea.value = lien;
        textArea.style.position = "fixed";
        textArea.style.top = "0";
        textArea.style.left = "0";
        textArea.style.opacity = "0";
        document.body.appendChild(textArea);
        textArea.focus();
        textArea.select();
        try {
            let successful = document.execCommand('copy');
            if (successful) {
                notifierSucces();
            } else {
                notifierErreur();
            }
        } catch (err) {
            console.error('Erreur Fallback :', err);
            notifierErreur();
        }
        document.body.removeChild(textArea);
    }
}

window.retourAuxCartes = function(event, fallbackUrl) {
    if (event) event.preventDefault();

    // Si on a navigué entre les onglets du profil via pushState,
    // history.back() ne ferait que revenir à l'onglet précédent :
    // on redirige donc directement vers la dernière page de cartes.
    const isProfileTabNavigation = Boolean(window.history.state && window.history.state.profileTabUrl);

    if (!isProfileTabNavigation && document.referrer && document.referrer.includes(window.location.host)) {
        let refUrl = new URL(document.referrer);
        let path = refUrl.pathname;

        let base = new URL(window.siteConfig.baseUrl).pathname;
        if (path.startsWith(base)) {
            path = '/' + path.substring(base.length);
        }
        path = path.replace(/\/\/+/g, '/');

        // Si la page EXACTE précédente est l'accueil (/) ou une catégorie (/c/1, /c/2...)
        // On fait un "retour arrière" natif pour conserver le défilement
        if (path === '/' || /^\/[pc]\/\d+$/.test(path)) {
            window.history.back();
            return false;
        }
    }

    let lastCards = sessionStorage.getItem('last_cards_url');
    window.location.href = lastCards ? lastCards : fallbackUrl;
    return false;
};

        async function processBatch(cards, batchSize = 5) {
            for (let i = 0; i < cards.length; i += batchSize) {
                const batch = cards.slice(i, i + batchSize);
                await Promise.all(batch.map(card => checkCardDispo(card, supportedDomains)));
                if (i + batchSize < cards.length) {
                    await new Promise(resolve => setTimeout(resolve, 150));
                }

        async function checkCardDispo(card, supportedDomains) {
            const itemId = card.getAttribute('data-id');
            const url = card.getAttribute('data-url');
            const statusDiv = document.getElementById(`live-status-${itemId}`);
            const dateContainer = document.getElementById(`date-container-${itemId}`);

            const isSupported = url && supportedDomains.some(domain => url.includes(domain));
            if (!isSupported) {
                if (statusDiv) statusDiv.style.display = 'none';
                return;
            }

            const cacheKey = `dispo_check_${itemId}_${url}`;
            try {
                const cachedResult = sessionStorage.getItem(cacheKey);
                if (cachedResult) {
                    const cacheData = JSON.parse(cachedResult);
                    const ageInMinutes = (Date.now() - cacheData.timestamp) / 60000;
                    if (ageInMinutes < 60) {
                        applyDispoResult(cacheData.disponible, statusDiv, dateContainer);
                        return;
                    }
                }
            } catch (_) {}

            try {
                const baseUrl = siteConfig.baseUrl.endsWith('/') ? siteConfig.baseUrl : siteConfig.baseUrl + '/';
                const response = await fetch(`${baseUrl}q?urlCible=${encodeURIComponent(url)}`, {
                    method: 'GET',
                    headers: { 'X-Requested-With': 'XMLHttpRequest' },
                    credentials: 'same-origin'
                });

                if (!response.ok) {
                    if (statusDiv) statusDiv.style.display = 'none';
                    return;
                }

                const data = await response.json();
                if (!data || !data.success) {
                    if (statusDiv) statusDiv.style.display = 'none';
                    return;
                }

                try {
                    sessionStorage.setItem(cacheKey, JSON.stringify({
                        timestamp: Date.now(),
                        disponible: Boolean(data.disponible)
                    }));
                } catch (_) {}

                applyDispoResult(Boolean(data.disponible), statusDiv, dateContainer);
            } catch (_) {
                if (statusDiv) statusDiv.style.display = 'none';
            }
        }

        function applyDispoResult(disponible, statusDiv, dateContainer) {
            if (statusDiv) statusDiv.style.display = 'none';
            if (!disponible && dateContainer) {
                dateContainer.style.display = 'block';
                if (dateContainer.innerHTML.trim() === '') {
                    dateContainer.innerHTML = `<p class="card-date" style="color: var(--danger);">Épisode non disponible.</p>`;
                }
            }
        }

        if (cardsToCheck.length > 0) {
            setTimeout(() => processBatch(cardsToCheck, 5), 500);
        }
    });
```

```php
<?php

declare(strict_types=1);

use CodeIgniter\Boot;
use Config\Paths;

/*
 *---------------------------------------------------------------
 * CHECK PHP VERSION
 *---------------------------------------------------------------
 */

$minPhpVersion = '8.1'; // If you update this, don't forget to update `spark`.
if (version_compare(PHP_VERSION, $minPhpVersion, '<')) {
    $message = sprintf(
        'Your PHP version must be %s or higher to run CodeIgniter. Current version: %s',
        $minPhpVersion,
        PHP_VERSION,
    );

    header('HTTP/1.1 503 Service Unavailable.', true, 503);
    echo $message;

    exit(1);
}

/*
 *---------------------------------------------------------------
 * SET THE CURRENT DIRECTORY
 *---------------------------------------------------------------
 */

// Path to the front controller (this file)
define('FCPATH', __DIR__.DIRECTORY_SEPARATOR);

// Ensure the current directory is pointing to the front controller's directory
if (FCPATH !== getcwd().DIRECTORY_SEPARATOR) {
    chdir(FCPATH);
}

/*
 *---------------------------------------------------------------
 * BOOTSTRAP THE APPLICATION
 *---------------------------------------------------------------
 * This process sets up the path constants, loads and registers
 * our autoloader, along with Composer's, loads our constants
 * and fires up an environment-specific bootstrapping.
 */

// LOAD OUR PATHS CONFIG FILE
// This is the line that might need to be changed, depending on your folder structure.
require FCPATH.'../app/Config/Paths.php';
// ^^^ Change this line if you move your application folder

$paths = new Paths();

// LOAD THE FRAMEWORK BOOTSTRAP FILE
require $paths->systemDirectory.'/Boot.php';

exit(Boot::bootWeb($paths));
```
